  process.env.ALLOWED_ORIGINS?.split(',') || []

  origin: (origin, cb) => {
    if (!origin || allowedOrigins.has(origin)) {
      cb(null, true);
    } else {
      cb(new Error('CORS Blocked'));
    }
  }
}));
